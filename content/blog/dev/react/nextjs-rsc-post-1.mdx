## Server Component vs Client Component

### Server Component (기본)

Next.js 15에서는 모든 컴포넌트가 기본적으로 Server Component입니다:

```tsx
// app/blog/page.tsx
import { getBlogPosts } from '@/lib/db';

export default async function BlogPage() {
  // 서버에서 직접 데이터 fetching
  const posts = await getBlogPosts();

  return (
    <div>
      <h1>Blog Posts</h1>
      {posts.map(post => (
        <article key={post.id}>
          <h2>{post.title}</h2>
          <p>{post.excerpt}</p>
        </article>
      ))}
    </div>
  );
}
```

### Client Component

`"use client"` 지시자가 필요한 경우:

```tsx
"use client";

import { useState, useEffect } from 'react';

export function LikeButton({ postId }: { postId: string }) {
  const [likes, setLikes] = useState(0);
  const [isLiked, setIsLiked] = useState(false);

  // ✅ useState, useEffect 사용 가능
  useEffect(() => {
    fetch(`/api/likes/${postId}`)
      .then(res => res.json())
      .then(data => setLikes(data.count));
  }, [postId]);

  // ✅ 이벤트 핸들러 사용 가능
  const handleClick = () => {
    setIsLiked(!isLiked);
    setLikes(isLiked ? likes - 1 : likes + 1);
  };

  return (
    <button onClick={handleClick}>
      {isLiked ? '❤️' : '🤍'} {likes}
    </button>
  );
}
```

## 컴포지션 패턴

Server와 Client Component를 효과적으로 조합:

```tsx
// app/dashboard/page.tsx (Server Component)
import { getUserData, getAnalytics } from '@/lib/db';
import { InteractiveChart } from './InteractiveChart'; // Client Component

export default async function DashboardPage() {
  // 서버에서 병렬로 데이터 fetch
  const [user, analytics] = await Promise.all([
    getUserData(),
    getAnalytics()
  ]);

  return (
    <div>
      {/* Server에서 렌더링 */}
      <header>
        <h1>Welcome, {user.name}</h1>
        <p>Last login: {user.lastLogin}</p>
      </header>

      {/* Client Component에 데이터 전달 */}
      <InteractiveChart data={analytics} />
    </div>
  );
}
```

```tsx
// app/dashboard/InteractiveChart.tsx (Client Component)
"use client";

import { useState } from 'react';
import { LineChart } from '@/components/charts';

export function InteractiveChart({ data }: { data: Analytics[] }) {
  const [timeRange, setTimeRange] = useState('7d');

  const filteredData = data.filter(item => {
    // 클라이언트에서 필터링
    return isWithinTimeRange(item.date, timeRange);
  });

  return (
    <div>
      <select value={timeRange} onChange={(e) => setTimeRange(e.target.value)}>
        <option value="7d">Last 7 days</option>
        <option value="30d">Last 30 days</option>
        <option value="90d">Last 90 days</option>
      </select>

      <LineChart data={filteredData} />
    </div>
  );
}
```

## 데이터 Fetching 패턴

### 1. 병렬 Fetching

```typescript
// app/product/[id]/page.tsx
async function getProduct(id: string) {
  const res = await fetch(`https://api.example.com/products/${id}`, {
    next: { revalidate: 3600 } // 1시간 캐싱
  });
  return res.json();
}

async function getReviews(id: string) {
  const res = await fetch(`https://api.example.com/reviews/${id}`);
  return res.json();
}

export default async function ProductPage({ params }: { params: { id: string } }) {
  // 병렬로 실행 - 더 빠름!
  const [product, reviews] = await Promise.all([
    getProduct(params.id),
    getReviews(params.id)
  ]);

  return (
    <div>
      <h1>{product.name}</h1>
      <p>${product.price}</p>

      <section>
        <h2>Reviews</h2>
        {reviews.map(review => (
          <Review key={review.id} data={review} />
        ))}
      </section>
    </div>
  );
}
```

### 2. Streaming with Suspense

```tsx
import { Suspense } from 'react';

// 느린 컴포넌트
async function SlowComponent() {
  const data = await fetchSlowData(); // 3초 걸림
  return <div>{data}</div>;
}

// 빠른 컴포넌트
async function FastComponent() {
  const data = await fetchFastData(); // 100ms
  return <div>{data}</div>;
}

export default function Page() {
  return (
    <div>
      {/* 빨리 로드됨 */}
      <FastComponent />

      {/* Suspense로 감싸서 나중에 렌더링 */}
      <Suspense fallback={<LoadingSkeleton />}>
        <SlowComponent />
      </Suspense>
    </div>
  );
}
```

## 주의사항

### ❌ 흔한 실수

```tsx
// ❌ Server Component에서 useState 사용
export default function Page() {
  const [count, setCount] = useState(0); // Error!
  return <div>{count}</div>;
}

// ❌ Server Component에서 이벤트 핸들러
export default function Page() {
  return <button onClick={() => alert('Hi')}>Click</button>; // Error!
}

// ❌ Client Component에서 async
"use client";
export default async function Page() { // Error!
  const data = await fetch('...');
  return <div>{data}</div>;
}
```

### ✅ 올바른 방법

```tsx
// ✅ Server Component에서 데이터 fetch 후 Client Component에 전달
export default async function Page() {
  const data = await fetch('...');

  return <ClientComponent data={data} />;
}

// ✅ Client Component는 interactivity만 담당
"use client";
export function ClientComponent({ data }) {
  const [selected, setSelected] = useState(data[0]);

  return (
    <div onClick={() => setSelected(...)}>
      {selected.name}
    </div>
  );
}
```

## 성능 최적화 팁

### 1. Server Component 우선

```tsx
// ✅ 가능한 한 Server Component로 유지
export default async function Layout({ children }) {
  const config = await getConfig(); // 서버에서만 실행

  return (
    <div>
      <StaticHeader config={config} />
      {children}
      <StaticFooter />
    </div>
  );
}
```

### 2. Client Component는 리프 노드에

```tsx
// ✅ 트리의 하단에 Client Component 배치
export default async function Page() {
  const data = await getData();

  return (
    <div> {/* Server */}
      <Header /> {/* Server */}
      <MainContent> {/* Server */}
        <InteractiveWidget data={data} /> {/* Client - 여기만! */}
      </MainContent>
    </div>
  );
}
```

### 3. 불필요한 클라이언트 번들 제거

```tsx
// ✅ Server Component에서 heavy 라이브러리 사용
import { marked } from 'marked'; // 번들에 포함 안됨!

export default async function BlogPost({ slug }) {
  const content = await getMarkdown(slug);
  const html = marked(content); // 서버에서 변환

  return <article dangerouslySetInnerHTML={{ __html: html }} />;
}
```

## 요약

| 이벤트 핸들러 | ❌ | ✅ |
