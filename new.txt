  17 │ 
  18 │   const handleResizeStart = (e: React.MouseEvent) => {
  19 │     e.preventDefault();
  20 │     e.stopPropagation();
  21 │     setIsResizing(true);
  22 │     resizeStartRef.current = {
  23 │       width: window.size?.width || 800,
  24 │       height: window.size?.height || 600,
  25 │       mouseX: e.clientX,
  26 │       mouseY: e.clientY,
  27 │     };
  28 │   };
  29 │ 
  30 │   React.useEffect(() => {
  31 │     if (!isResizing) return;
  32 │ 
  33 │     const handleResizeMove = (e: MouseEvent) => {
  34 │       if (!resizeStartRef.current) return;
  35 │ 
  36 │       const deltaX = e.clientX - resizeStartRef.current.mouseX;
  37 │       const deltaY = e.clientY - resizeStartRef.current.mouseY;
  38 │ 
  39 │       const newWidth = Math.max(400, resizeStartRef.current.width + deltaX);
  40 │       const newHeight = Math.max(300, resizeStartRef.current.height + deltaY);
  41 │ 
  42 │       updateWindowSize(window.id, { width: newWidth, height: newHeight });
  43 │     };
  44 │ 
  45 │     const handleResizeEnd = () => {
  46 │       setIsResizing(false);
  47 │       resizeStartRef.current = null;
  48 │     };
  49 │ 
  50 │     globalThis.window.addEventListener('mousemove', handleResizeMove);
  51 │     globalThis.window.addEventListener('mouseup', handleResizeEnd);
  52 │ 
  53 │     return () => {
  54 │       globalThis.window.removeEventListener('mousemove', handleResizeMove);
  55 │       globalThis.window.removeEventListener('mouseup', handleResizeEnd);
  56 │     };
  57 │   }, [isResizing, window.id, updateWindowSize]);

  23 │ export function Sidebar({ posts, currentPath, onNavigate, selectedPath, rootPath = [] }: SidebarProps) {
  24 │   // Build tree from posts
  25 │   const tree: Record<string, TreeNode> = {};
  26 │ 
  27 │   posts.forEach((post) => {
  28 │     let currentLevel = tree;
  29 │     let currentPathAccum: string[] = [];
  30 │ 
  31 │     // Filter posts: Only include if they start with rootPath
  32 │     if (rootPath.length > 0) {
  33 │       const postPathStr = post.path.join('/');
  34 │       const rootPathStr = rootPath.join('/');
  35 │       if (!postPathStr.startsWith(rootPathStr)) return;
  36 │     }
  37 │ 
  38 │     post.path.forEach((part, index) => {
  39 │       currentPathAccum = [...currentPathAccum, part];
  40 │ 
  41 │       if (!currentLevel[part]) {
  42 │         currentLevel[part] = {
  43 │           name: part,
  44 │           path: [...currentPathAccum],
  45 │           children: {},
  46 │         };
  47 │       }
  48 │ 
  49 │       // If we are deeper, move reference
  50 │       if (index < post.path.length) {
  51 │         currentLevel = currentLevel[part].children!;
  52 │       }
  53 │     });
  54 │   });
  55 │ 
  56 │   // Calculate the starting node for rendering
  57 │   // If rootPath is ['ai'], we want to render the children of 'ai', not 'ai' itself.

  59 │   let displayRoot = tree;
  60 │   if (rootPath.length > 0) {
  61 │     // Traverse down to root
  62 │     for (const part of rootPath) {
  63 │       if (displayRoot[part] && displayRoot[part].children) {
  64 │         displayRoot = displayRoot[part].children!;
  65 │       } else {
  66 │         displayRoot = {}; // Path not found or empty
  67 │         break;
  68 │       }
  69 │     }
  70 │   }
  71 │

  23 │   const list = fs.readdirSync(dir);
  24 │

  31 │     } else {
  32 │       if (file.endsWith('.mdx')) {
  33 │         results.push(filePath);
  34 │       }
  35 │     }

  37 │
